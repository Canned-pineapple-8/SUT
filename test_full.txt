kotlin_prog Example {
    def {
        var x, y: Int
        var temp: Float
        var isActive: Boolean
        data_class Person(
            age: Int,
            salary: Float
        )
        var p: Person
    }

    fun_main() {
        x = (10 + 5) * 3 - 2  // 43
        temp = 3.14e14 * 100
        isActive = true

        p.age = 25
        p.salary = temp + 500.0

        if (x >= 40 && !isActive || p.age == 25) {
            y = x / 2
            p.salary = p.salary * 1.1
        } else {
            y = 0
        }
        // Комментарий в конце
    }
}
